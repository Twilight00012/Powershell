$files = Get-ChildItem -Path "C:\day5" -File
$extensions = $files.Extension | Sort-Object -Unique

foreach ($ext in $extensions) {
    $folderName = $ext.TrimStart(".")
    $folderPath = Join-Path -Path "C:\day5" -ChildPath $folderName

    if (-not (Test-Path -Path $folderPath)) {
        New-Item -ItemType Directory -Path $folderPath
    }
}
foreach ($file in $files) {
    $folderName = $file.Extension.TrimStart(".")
    $folderPath = Join-Path -Path "C:\day5" -ChildPath $folderName
    Move-Item -Path $file.FullName -Destination $folderPath
}
